# ==================== 任务调度配置 ====================
# 待处理任务分发间隔（分钟）
# 建议值：批量任务 5-10 分钟，日常任务 30 分钟
DISPATCH_INTERVAL_MINUTES=5

# 失败任务重试间隔（分钟）
RETRY_INTERVAL_MINUTES=30

# 最大重试次数
MAX_RETRY_COUNT=3

# 每批重试的任务数量
RETRY_BATCH_SIZE=100

# ==================== 智能重试配置 ====================
# 失败后至少等待多久才重试（分钟）
# 建议值：预计全量任务耗时 × 0.8
RETRY_MIN_WAIT_MINUTES=60

# 如果 pending 任务超过此数量，跳过失败任务重试
# 建议值：总任务数 × 0.1
RETRY_SKIP_IF_PENDING_OVER=100

# ==================== 进程池配置 ====================
# 最大工作进程数（同时执行的任务数）
# 建议值：低配机器 2-3，中配 3-5，高配 5-10
MAX_WORKERS=5

# 工作进程内存限制（MB）
WORKER_MEMORY_LIMIT_MB=512
